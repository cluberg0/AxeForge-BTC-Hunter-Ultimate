// Native Apple SF Pro + dynamically resolved iOS colors.
// The same SwiftUI views render in system, light and dark modes.
private enum Ink {
    private static func adaptive(_ light: (CGFloat, CGFloat, CGFloat),
                                 _ dark: (CGFloat, CGFloat, CGFloat),
                                 alpha: CGFloat = 1) -> Color {
        Color(uiColor: UIColor { traits in
            let rgb = traits.userInterfaceStyle == .dark ? dark : light
            return UIColor(red: rgb.0, green: rgb.1, blue: rgb.2, alpha: alpha)
        })
    }
    static let bg = adaptive((0.89, 0.94, 0.98), (0.025, 0.038, 0.069))
    static let surface = adaptive((0.985, 0.990, 1.000), (0.078, 0.107, 0.169))
    static let surface2 = adaptive((0.876, 0.919, 0.959), (0.122, 0.171, 0.246))
    static let cyan = adaptive((0.068, 0.48, 0.67), (0.43, 0.89, 0.95))
    static let violet = adaptive((0.35, 0.30, 0.76), (0.55, 0.49, 0.98))
    static let faint = adaptive((0.34, 0.42, 0.51), (0.65, 0.72, 0.82))
    static let text = adaptive((0.066, 0.122, 0.205), (0.98, 0.99, 1.00))
    static let onAccent = adaptive((1, 1, 1), (0.030, 0.043, 0.071))
    static let border = adaptive((0.50, 0.61, 0.73), (0.83, 0.90, 0.99), alpha: 0.16)
}

private struct InkCard<Content: View>: View {
    let content: Content
    init(@ViewBuilder content: () -> Content) { self.content = content() }
    var body: some View {
        content
            .padding(16)
            .frame(maxWidth: .infinity, alignment: .leading)
            .background(RoundedRectangle(cornerRadius: 19, style: .continuous)
                .fill(.ultraThinMaterial))
            .background(RoundedRectangle(cornerRadius: 19, style: .continuous)
                .fill(LinearGradient(colors: [Ink.surface.opacity(0.45), Ink.surface2.opacity(0.13)],
                                      startPoint: .topLeading, endPoint: .bottomTrailing)))
            .overlay(RoundedRectangle(cornerRadius: 19, style: .continuous)
                .strokeBorder(Ink.border.opacity(0.9), lineWidth: 0.7))
            .shadow(color: .black.opacity(0.07), radius: 14, y: 5)
    }
}

    private var boardTasks: [X1BoardTask] {
        guard let data = boardJSON.data(using: .utf8),
              let items = try? JSONDecoder().decode([X1BoardTask].self, from: data) else { return [] }
        return items
    }

    private func storeBoardTasks(_ tasks: [X1BoardTask]) {
        guard let data = try? JSONEncoder().encode(tasks),
              let json = String(data: data, encoding: .utf8) else { return }
        boardJSON = json
    }

    private func createBoardTask() {
        let title = newBoardTitle.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !title.isEmpty else { return }
        var tasks = boardTasks
        tasks.insert(X1BoardTask(title: title), at: 0)
        storeBoardTasks(tasks)
        newBoardTitle = ""
    }

    private func advanceTask(_ id: UUID) {
        var tasks = boardTasks
        guard let i = tasks.firstIndex(where: { $0.id == id }) else { return }
        let lanes = ["В работе", "Проверка", "Готово"]
        let current = lanes.firstIndex(of: tasks[i].lane) ?? 0
        tasks[i].lane = lanes[min(current + 1, lanes.count - 1)]
        storeBoardTasks(tasks)
    }

    private func deleteBoardTask(_ id: UUID) {
        storeBoardTasks(boardTasks.filter { $0.id != id })
    }

    private func plotLine(_ series: [Double], color: Color) -> some View {
        GeometryReader { geometry in
            Path { path in
                guard !series.isEmpty else { return }
                let height = geometry.size.height
                let width = geometry.size.width
                let peak = max(1, series.max() ?? 1)
                for (index, sample) in series.enumerated() {
                    let x = width * CGFloat(index) / CGFloat(max(1, series.count - 1))
                    let y = height - height * CGFloat(max(0, sample) / peak)
                    if index == 0 { path.move(to: CGPoint(x: x, y: y)) }
                    else { path.addLine(to: CGPoint(x: x, y: y)) }
                }
            }
            .stroke(color, style: StrokeStyle(lineWidth: 2.5, lineCap: .round, lineJoin: .round))
        }
        .frame(height: 100)
        .accessibilityLabel("График нагрузки за последние измерения")
    }

    private var canSend: Bool {
        !draft.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty
        && !llamaState.activeModelName.isEmpty
        && !llamaState.isGenerating && !llamaState.loading
    }

    private func sendMessage() {
        guard canSend else { return }
        let input = draft.trimmingCharacters(in: .whitespacesAndNewlines)
        if !llamaState.generatedText.isEmpty {
            history.append(ChatEntry(role: .assistant, text: llamaState.generatedText))
        }
        llamaState.generatedText = ""
        history.append(ChatEntry(role: .user, text: input))
        persistChat()
        draft = ""
        composerFocused = false
        let payload = attachedText.isEmpty ? input :
            "[Файл: \(attachedName)]\n\(attachedText.prefix(1700))\n\nВопрос: \(input)"
        attachedText = ""
        attachedName = ""
        if speech.listening { speech.stopListening() }
        var turns: [X1Turn] = []
        if !systemPrompt.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty {
            turns.append(X1Turn(role: "system", content: systemPrompt))
        }
        turns.append(contentsOf: history.suffix(24).map { message in
            X1Turn(role: message.role == .user ? "user" : "assistant", content: message.text)
        })
        if let last = turns.indices.last {
            turns[last] = X1Turn(role: "user", content: payload)
        }
        Task { await llamaState.complete(turns: turns) }
    }


    // Offline history: stored only under Documents/x1er_ai/chat_history.json.
    private func restoreChatArchive() {
        archivedChats = X1ChatArchive.load()
        if let latest = archivedChats.first {
            activeChatID = latest.id
            history = latest.messages.map { item in
                ChatEntry(role: item.role == "user" ? .user : .assistant, text: item.text)
            }
            llamaState.generatedText = ""
        }
    }

    private func persistChat() {
        var messages = history.map { item in
            X1StoredMessage(id: item.id, role: item.role == .user ? "user" : "assistant", text: item.text)
        }
        if !llamaState.generatedText.isEmpty {
            messages.append(X1StoredMessage(id: UUID(), role: "assistant", text: llamaState.generatedText))
        }
        guard !messages.isEmpty else { return }
        let lead = messages.first(where: { $0.role == "user" })?.text ?? "Новый диалог"
        let firstLine = String(lead.split(separator: "\n").first ?? "Новый диалог")
        let title = String(firstLine.prefix(47))
        let chat = X1StoredChat(id: activeChatID,
                                title: title.isEmpty ? "Новый диалог" : title,
                                updatedAt: Date(), messages: messages)
        if let idx = archivedChats.firstIndex(where: { $0.id == activeChatID }) {
            archivedChats[idx] = chat
        } else {
            archivedChats.insert(chat, at: 0)
        }
        do {
            try X1ChatArchive.save(archivedChats)
            archiveError = ""
        } catch {
            archiveError = "Не удалось сохранить историю: \(error.localizedDescription)"
        }
    }

    private func newChat() {
        guard !llamaState.isGenerating && !llamaState.loading else { return }
        persistChat()
        activeChatID = UUID()
        history = []
        llamaState.generatedText = ""
        Task { await llamaState.clear() }
    }

    private func openChat(_ chat: X1StoredChat) {
        guard !llamaState.isGenerating && !llamaState.loading else { return }
        persistChat()
        activeChatID = chat.id
        history = chat.messages.map {
            ChatEntry(role: $0.role == "user" ? .user : .assistant, text: $0.text)
        }
        llamaState.generatedText = ""
        Task { await llamaState.clear() }
        showChatHistory = false
    }

    private func eraseChat(_ chat: X1StoredChat) {
        archivedChats.removeAll { $0.id == chat.id }
        if activeChatID == chat.id {
            activeChatID = UUID()
            history = []
            llamaState.generatedText = ""
            Task { await llamaState.clear() }
        }
        do {
            try X1ChatArchive.save(archivedChats)
        } catch {
            archiveError = error.localizedDescription
        }
    }

        }

    private func importGGUF(_ source: URL) {
        guard !importing else { return }
        importing = true
        let dest = library.modelsDirectory.appendingPathComponent(source.lastPathComponent)
        Task.detached(priority: .userInitiated) {
            let access = source.startAccessingSecurityScopedResource()
            defer { if access { source.stopAccessingSecurityScopedResource() } }
            do {
                guard source.pathExtension.lowercased() == "gguf" else {
                    throw NSError(domain: "GGUF", code: -1,
                                  userInfo: [NSLocalizedDescriptionKey: "Нужен файл .gguf"])
                }
                if source.standardizedFileURL != dest.standardizedFileURL,
                   !FileManager.default.fileExists(atPath: dest.path) {
                    try FileManager.default.copyItem(at: source, to: dest)
                }
                await MainActor.run { library.refreshLocal(); importing = false }
            } catch {
                await MainActor.run {
                    importError = error.localizedDescription
                    importing = false
                }
            }
        }
    }
