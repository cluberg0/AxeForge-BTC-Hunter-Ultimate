import SwiftUI
import UniformTypeIdentifiers

private enum Ink {
    static let bg = Color(red: 0.035, green: 0.052, blue: 0.085)
    static let surface = Color(red: 0.085, green: 0.111, blue: 0.158)
    static let surface2 = Color(red: 0.112, green: 0.147, blue: 0.197)
    static let cyan = Color(red: 0.38, green: 0.89, blue: 0.94)
    static let violet = Color(red: 0.55, green: 0.49, blue: 0.98)
    static let faint = Color(red: 0.62, green: 0.67, blue: 0.77)
    static let border = Color.white.opacity(0.09)
}

private struct InkCard<Content: View>: View {
    let content: Content
    init(@ViewBuilder content: () -> Content) { self.content = content() }
    var body: some View {
        content
            .padding(17)
            .frame(maxWidth: .infinity, alignment: .leading)
            .background(Ink.surface, in: RoundedRectangle(cornerRadius: 22, style: .continuous))
            .overlay(RoundedRectangle(cornerRadius: 22, style: .continuous)
                .stroke(Ink.border, lineWidth: 1))
    }
}

private enum RunnerPage: String, CaseIterable {
    case chat = "Чат"
    case library = "Модели"
    case settings = "Настройки"
    var icon: String {
        switch self {
        case .chat: return "bubble.left.and.bubble.right.fill"
        case .library: return "square.stack.3d.up.fill"
        case .settings: return "slider.horizontal.3"
        }
    }
}

private struct ChatEntry: Identifiable {
    enum Role { case user, assistant }
    let id = UUID()
    let role: Role
    let text: String
}

struct ContentView: View {
    @StateObject private var llamaState = LlamaState()
    @StateObject private var library = HFModelLibrary()
    @AppStorage("inkling_use_gpu") private var metalEnabled = false

    @State private var page: RunnerPage = .chat
    @State private var draft = ""
    @State private var history: [ChatEntry] = []
    @State private var showImporter = false
    @State private var importing = false
    @State private var importError = ""
    @State private var deleteTarget: URL?
    @FocusState private var composerFocused: Bool

    var body: some View {
        ZStack {
            Ink.bg.ignoresSafeArea()
            RadialGradient(colors: [Ink.violet.opacity(0.12), .clear],
                           center: .topTrailing, startRadius: 10, endRadius: 370)
                .ignoresSafeArea()
            VStack(spacing: 0) {
                header
                Rectangle().fill(Ink.border).frame(height: 1)
                Group {
                    switch page {
                    case .chat: chatView
                    case .library: libraryView
                    case .settings: settingsView
                    }
                }
                bottomNavigation
            }
        }
        .tint(Ink.cyan)
        .preferredColorScheme(.dark)
        .fileImporter(isPresented: $showImporter,
                      allowedContentTypes: [UTType(filenameExtension: "gguf") ?? .data],
                      allowsMultipleSelection: false) { selection in
            switch selection {
            case .success(let urls):
                if let selected = urls.first { importGGUF(selected) }
            case .failure(let error):
                importError = error.localizedDescription
            }
        }
        .alert("Ошибка импорта", isPresented: Binding(
            get: { !importError.isEmpty },
            set: { if !$0 { importError = "" } }
        )) { Button("Понятно", role: .cancel) { importError = "" } } message: {
            Text(importError)
        }
        .confirmationDialog("Удалить эту GGUF-модель с телефона?",
                            isPresented: Binding(get: { deleteTarget != nil },
                                                set: { if !$0 { deleteTarget = nil } }),
                            titleVisibility: .visible) {
            Button("Удалить файл", role: .destructive) {
                if let target = deleteTarget { library.deleteModel(target) }
                deleteTarget = nil
            }
            Button("Отмена", role: .cancel) { deleteTarget = nil }
        }
        .onAppear { library.refreshLocal() }
    }

                    }
                    .padding(.horizontal, 18).padding(.vertical, 18)

                .onChange(of: llamaState.generatedText) { _ in
                    withAnimation(.easeOut(duration: 0.15)) { proxy.scrollTo("chatEnd", anchor: .bottom) }
                }
            }
            composer
        }
    }

    private func bubble(text: String, user: Bool) -> some View {
        HStack {
            if user { Spacer(minLength: 38) }
            VStack(alignment: .leading, spacing: 7) {
                Text(user ? "ТЫ" : "INKLING")
                    .font(.system(size: 9, weight: .bold, design: .monospaced))
                    .tracking(1.6).foregroundStyle(user ? Ink.cyan : Ink.violet)
                Text(text).font(.system(size: 14))
                    .foregroundStyle(.white.opacity(0.94))
                    .textSelection(.enabled)
            }
            .padding(15)
            .background(user ? Ink.surface2 : Ink.surface,
                        in: RoundedRectangle(cornerRadius: 17))
            .overlay(RoundedRectangle(cornerRadius: 17).stroke(Ink.border, lineWidth: 1))
            if !user { Spacer(minLength: 20) }
        }
    }

    private var composer: some View {
        HStack(alignment: .bottom, spacing: 10) {
            TextField("Спроси что-нибудь...", text: $draft, axis: .vertical)
                .lineLimit(1...4)
                .focused($composerFocused)
                .font(.system(size: 14))
                .padding(.horizontal, 14).padding(.vertical, 13)
                .background(Ink.surface2, in: RoundedRectangle(cornerRadius: 17))
            Button(action: sendMessage) {
                Image(systemName: "arrow.up")
                    .font(.system(size: 17, weight: .bold))
                    .foregroundStyle(Ink.bg)
                    .frame(width: 46, height: 46)
                    .background(canSend ? Ink.cyan : Ink.faint,
                                in: RoundedRectangle(cornerRadius: 15))
            }
            .disabled(!canSend)
        }
        .padding(.horizontal, 17).padding(.vertical, 12)
        .background(Ink.bg)
    }

    private var canSend: Bool {
        !draft.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty
        && !llamaState.activeModelName.isEmpty
        && !llamaState.isGenerating && !llamaState.loading
    }

    private func sendMessage() {
        guard canSend else { return }
        let input = draft.trimmingCharacters(in: .whitespacesAndNewlines)
        if !llamaState.generatedText.isEmpty {
            history.append(ChatEntry(role: .assistant, text: llamaState.generatedText))
        }
        history.append(ChatEntry(role: .user, text: input))
        draft = ""
        composerFocused = false
        Task { await llamaState.complete(text: input) }
    }

            Spacer()
        }
        .font(.system(size: 15, weight: .bold, design: .rounded))
        .padding(.horizontal, 3)
    }

    private func titleBlock(_ title: String, subtitle: String) -> some View {
        VStack(alignment: .leading, spacing: 5) {
            Text(title).font(.system(size: 27, weight: .bold, design: .rounded))
                .foregroundStyle(.white)
            Text(subtitle).font(.system(size: 12)).foregroundStyle(Ink.faint)
        }
        .padding(.horizontal, 3).padding(.bottom, 2)
    }

    private func importGGUF(_ source: URL) {
        guard !importing else { return }
        importing = true
        let dest = library.modelsDirectory.appendingPathComponent(source.lastPathComponent)
        Task.detached(priority: .userInitiated) {
            let access = source.startAccessingSecurityScopedResource()
            defer { if access { source.stopAccessingSecurityScopedResource() } }
            do {
                guard source.pathExtension.lowercased() == "gguf" else {
                    throw NSError(domain: "GGUF", code: -1,
                                  userInfo: [NSLocalizedDescriptionKey: "Нужен файл .gguf"])
                }
                if source.standardizedFileURL != dest.standardizedFileURL,
                   !FileManager.default.fileExists(atPath: dest.path) {
                    try FileManager.default.copyItem(at: source, to: dest)
                }
                await MainActor.run { library.refreshLocal(); importing = false }
            } catch {
                await MainActor.run {
                    importError = error.localizedDescription
                    importing = false
                }
            }
        }
    }
